sequence restart;
	clear;
	dialog "The next heir comes along...";
	branch start;

end_sequence;


sequence start;
	set_print_sound resource Adventure/PrintRandom;

	flag remove Weakness;
	flag remove Unarmored;
	flag remove GoblinsPursuing;
	flag remove EnchantedBoots;
	flag remove LightPotion2;
	flag remove HealingPotion2;
	flag remove MagicPotion2;

	font face DOS;
	font color name 255 255 0;

	background Adventure/1 crossfade 0.5;
	character_name Intro;
	dialog "Your father was an explorer," 0.2 " dungeon delver," 0.2 " and mage of great renown.";
	dialog "With his status and money, he left a trail of children across the land.";
	background Adventure/2 crossfade 0.2;
	dialog "That trail became a constant line of deaths," 0.2 " starting with the oldest and working onwards."; 
	dialog "They all passed away on their 20th birthday, " 0.2 "each from an accident or bad luck.";
	dialog "Eventually, " 0.2 "a pattern started being noticed.";
	dialog "Stories say that he was harvesting you for power," 0.2 " but even when he passed away," 0.2 " the trend continued.";
	dialog "You are next in line, having only a few days before your predetermined death.";
	dialog "In your research of curses," 0.2 " bargains," 0.2 " and stolen power," 0.2 " you believe you should be able to collect your father's heart and take it to an enchanter for cleansing."; 
	dialog "Hopefully," 0.2 " that will save you and your kin.";
	dialog "It won't be easy." 0.2 " Whatever pact he made entombed his body in a dungeon that none have been able to break into.";
	dialog "You believe your lineage should allow you entry.";
	dialog "Growing up," 0.2 " you studied in the art of...";

	branch choose_class;

end_sequence;

sequence choose_class;

	# Class selection;
	flag remove Spellcaster;flag remove Thief;flag remove Fighter;
		choice "Magic" flag add Spellcaster;
		choice "Stealth" flag add Thief;
		choice "Combat" flag add Fighter;
		end_choices;

	if flag Spellcaster dialog "You are a spellcaster.";
	if flag Thief dialog "You are a thief.";
	if flag Fighter dialog "You are a fighter.";
		choice "That is correct." pass;
		choice "That doesn't sound right" branch select_class_again;
		end_choices;

	character_name "You";
	dialog "Huzzah!";

	character_name "TIP";
	if flag Spellcaster dialog "You can cast spells to get out of trouble.";
	if flag Thief dialog "Your light armour and experience allows you the ability to sneak around without detection.";
	if flag Fighter dialog "Your combat ability and endurance are unrivaled and you have the equipment to match.";

	character_name "Choose a potion";
	dialog "And before leaving for the dungeon," 0.2 " you stocked up on supplies and found some extra money with which to buy a potion...";
	branch choose_potion;

end_sequence;

sequence select_class_again;
	character_name "Select your class";
	dialog "In that case, you are a...";
	branch choose_class;
end_sequence;


sequence choose_potion;

	flag remove HealingPotion;flag remove LightPotion;flag remove MagicPotion;

		choice "Potion of healing" flag add HealingPotion;
		choice "Potion of light" flag add LightPotion;
		choice "Potion of magic" flag add MagicPotion;
		end_choices;

	if flag HealingPotion dialog "You decided to bring with you a potion of healing.";
	if flag LightPotion dialog "You decided to bring with you a potion of light.";
	if flag MagicPotion dialog "You decided to bring with you a potion of magic.";

	choice "That is correct." branch Adventure/Basin start;
	choice "That doesn't sound right" branch choose_potion_again;

end_sequence;

sequence choose_potion_again;
	character_name "Choose a potion";
	dialog "In that case...";
	branch choose_potion;
end_sequence;
